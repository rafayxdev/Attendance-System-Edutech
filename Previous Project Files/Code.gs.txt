function setupUsersSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Users");
  if (!sheet) {
    sheet = ss.insertSheet("Users");
    var headers = ["Email", "PlainPassword", "PasswordHash", "Role", "Unique ID", "FullName", "Active"];
    sheet.getRange(1, 1, 1, headers.length).setValues([headers])
         .setFontWeight("bold").setBackground("#1e3a8a").setFontColor("white");
    sheet.setFrozenRows(1);
  }
}

function generateUsersFromEmailSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var emailSheet = ss.getSheetByName("EmailsUser");
  
  setupUsersSheet();
  var usersSheet = ss.getSheetByName("Users");

  // Format Headers
  emailSheet.getRange(1, 1, 1, 5).setValues([["Gmail", "Generated", "Role", "Name", "Unique ID"]]);
  usersSheet.getRange(1, 1, 1, 7).setValues([["Email", "PlainPassword", "PasswordHash", "Role", "Unique ID", "FullName", "Active"]]);

  var lastRow = emailSheet.getLastRow();
  if (lastRow < 2) return;

  var data = emailSheet.getRange(2, 1, lastRow - 1, 5).getValues();
  for (var i = 0; i < data.length; i++) {
    var email     = (data[i][0] || "").toString().trim().toLowerCase();
    var generated = (data[i][1] || "").toString().trim().toUpperCase();
    var role      = (data[i][2] || "Student").toString().trim();
    var name      = (data[i][3] || email.split("@")[0]).toString().trim();
    var uniqueId  = (data[i][4] || "N/A").toString().trim();

    if (!email || generated === "TRUE" || !email.endsWith("@gmail.com")) continue;

    var emailPrefix  = email.split("@")[0].substring(0, 6);
    var randomDigits = Math.floor(1000 + Math.random() * 9000).toString();
    var plainPass    = emailPrefix + "@" + randomDigits;
    
    if (role.toLowerCase() === "admin") {
      var adminDigits = Math.floor(100000 + Math.random() * 900000).toString();
      plainPass = emailPrefix + "#Admin@" + adminDigits;
    }

    var passHash = Utilities.base64Encode(plainPass);
    usersSheet.appendRow([email, plainPass, passHash, role, uniqueId, name, true]);
    emailSheet.getRange(i + 2, 2).setValue("TRUE");
  }
}

function findUser(email) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Users");
  if (!sheet) return null;
  var values = sheet.getDataRange().getValues();
  for (var i = 1; i < values.length; i++) {
    if ((values[i][0] || "").toLowerCase().trim() === email.toLowerCase().trim()) {
      return { 
        email: values[i][0], plainPass: values[i][1], passHash: values[i][2], 
        role: values[i][3], uniqueId: values[i][4], fullName: values[i][5], active: values[i][6] 
      };
    }
  }
  return null;
}

function loginUser(email, password, roleTab) {
  if (!email || !email.endsWith('@gmail.com')) 
    return { success: false, message: "Only Gmail addresses are allowed." };
  
  var user = findUser(email);
  if (!user) 
    return { success: false, message: "Email not registered in the system." };
  if (user.active !== true && user.active !== "TRUE" && user.active !== "true") 
    return { success: false, message: "Account deactivated." };

  var userRole = (user.role || "").toString().trim().toLowerCase();

  if (roleTab === 'admin' && userRole !== 'admin') 
    return { success: false, message: "You do not have admin access." };

  if (roleTab === 'user') {
    var allowedUsers = ['student', 'internee', 'faculty member', 'faculty', 'human resource', 'chief executive', 'employee'];
    if (allowedUsers.indexOf(userRole) === -1) {
      if (userRole === 'admin') 
        return { success: false, message: "Please use the Admin tab to log in." };
      return { success: false, message: "Role not recognized for User login." };
    }
  }

  if (password.trim() !== (user.plainPass || "").toString().trim()) 
    return { success: false, message: "Incorrect password." };

  // ← No token, just return user info directly
  return { 
    success: true, 
    role: userRole, 
    name: user.fullName, 
    email: user.email, 
    uniqueId: user.uniqueId 
  };
}

function processForm(data) {
  // ✅ Look up user directly by email sent from the form
  var user = findUser(data.email);
  if (!user) return "User account not found.";

  var sheet = getOrCreateSheet();
  var now = new Date(), tz = Session.getScriptTimeZone();
  var todayStr = Utilities.formatDate(now, tz, "yyyy-MM-dd");
  var timeVal = (now.getHours() * 100) + now.getMinutes();
  var values = sheet.getDataRange().getValues();

  var submitName     = user.fullName;
  var submitEmail    = user.email;
  var submitCategory = user.role;
  var submitUniqueId = user.uniqueId;

  // 1. Duplicate Check
  for (var i = 1; i < values.length; i++) {
    var rowDate = Utilities.formatDate(new Date(values[i][0]), tz, "yyyy-MM-dd");
    if (values[i][2] === submitName && rowDate === todayStr && values[i][5] === data.type) {
      return "You have already marked " + data.type + " for today.";
    }
  }

  // 2. Checkout Restriction
  var restrictedCats = ["Internee", "Employee", "Faculty", "Faculty Member", "Human Resource"];
  if (data.type === "Time Out" && restrictedCats.indexOf(submitCategory) !== -1) {
    if (timeVal < 1430) return "Early checkout is not allowed. Time Out only permitted after 2:30 PM.";
  }

  // 3. Time In Required
  var hasTimeIn = false;
  for (var j = 1; j < values.length; j++) {
    var rowDate2 = Utilities.formatDate(new Date(values[j][0]), tz, "yyyy-MM-dd");
    if (values[j][2] === submitName && rowDate2 === todayStr && values[j][5] === "Time In") {
      hasTimeIn = true; break;
    }
  }
  if (data.type === "Time Out" && !hasTimeIn) return "You must mark Time In before Time Out.";

  // 4. Late Logic
  var status = "On Time";
  if (data.type === "Time In") {
    var cat = submitCategory.toLowerCase();
    if ((cat === "employee" || cat.includes("faculty") || cat === "human resource") && timeVal > 915) status = "Late";
    else if (cat === "internee" && timeVal > 945) status = "Late";
  }

  // 5. Append
  sheet.appendRow([now, submitUniqueId, submitName, submitCategory, data.purpose || "N/A", data.type, data.location, submitEmail, status, data.ip || "N/A"]);

  if (submitEmail && submitEmail.trim() !== "") {
    var emailData = { name: submitName, type: data.type, location: data.location, category: submitCategory, purpose: data.purpose || "N/A" };
    try { sendConfirmationEmail(submitEmail, emailData, now, status, submitUniqueId); } catch (e) {}
  }

  updateMonthlySheet();
  return "Success";
}
function sendConfirmationEmail(recipientEmail, data, timestamp, status, uniqueId) {
  var template = HtmlService.createTemplateFromFile('EmailTemplate');
  template.name = data.name; template.type = data.type; template.location = data.location;
  template.status = status; template.uniqueId = uniqueId || "N/A";
  template.category = data.category || ""; template.purpose = data.purpose || "N/A";
  template.date = Utilities.formatDate(timestamp, Session.getScriptTimeZone(), "MMMM dd, yyyy");
  template.time = Utilities.formatDate(timestamp, Session.getScriptTimeZone(), "hh:mm a");

  MailApp.sendEmail({ to: recipientEmail, subject: "Attendance Receipt - " + data.name, htmlBody: template.evaluate().getContent() });
}

function processGuestForm(data) {
  var sheet = getOrCreateSheet();
  var now = new Date(), tz = Session.getScriptTimeZone();
  var todayStr = Utilities.formatDate(now, tz, "yyyy-MM-dd");
  var values = sheet.getDataRange().getValues();

  var submitName = data.name.trim();
  if (!submitName) return "Name is required.";

  // 1. Duplicate Check
  for (var i = 1; i < values.length; i++) {
    var rowDate = Utilities.formatDate(new Date(values[i][0]), tz, "yyyy-MM-dd");
    if (values[i][2].toString().trim().toLowerCase() === submitName.toLowerCase() && rowDate === todayStr && values[i][5] === data.type) {
      return "You have already marked " + data.type + " for today.";
    }
  }

  // 2. Time In Required Check
  var hasTimeIn = false;
  for (var j = 1; j < values.length; j++) {
    var rowDate2 = Utilities.formatDate(new Date(values[j][0]), tz, "yyyy-MM-dd");
    if (values[j][2].toString().trim().toLowerCase() === submitName.toLowerCase() && rowDate2 === todayStr && values[j][5] === "Time In") {
      hasTimeIn = true; break;
    }
  }
  if (data.type === "Time Out" && !hasTimeIn) return "You must mark Time In before marking Time Out.";

  var status = "On Time"; // Guests are always considered On Time
  var category = "Guest";
  var uniqueId = "N/A";

  sheet.appendRow([now, uniqueId, submitName, category, data.purpose || "N/A", data.type, data.location, data.email || "N/A", status, data.ip || "Unknown"]);

  if (data.email && data.email.trim() !== "") {
    var emailData = { name: submitName, type: data.type, location: data.location, category: category, purpose: data.purpose || "N/A" };
    try { sendConfirmationEmail(data.email, emailData, now, status, uniqueId); } catch (e) {}
  }

  updateMonthlySheet();
  return "Success";
}

function getAdminStats() {
  // var td = validateToken(token);
  // if (!td || td.role.toLowerCase() !== 'admin') return null;
  
  var values = getOrCreateSheet().getDataRange().getValues(), tz = Session.getScriptTimeZone();
  var todayStr = Utilities.formatDate(new Date(), tz, "yyyy-MM-dd");
  var present = 0, late = 0, guests = 0, timeouts = 0, countedNames = [];

  for (var i = 1; i < values.length; i++) {
    if (!values[i][0] || !values[i][2]) continue;
    if (Utilities.formatDate(new Date(values[i][0]), tz, "yyyy-MM-dd") !== todayStr) continue;

    var name = values[i][2], type = (values[i][5] || "").toString().trim(), status = (values[i][8] || "").toString().trim().toUpperCase();

    if (type === "Time In" && countedNames.indexOf(name) === -1) { present++; countedNames.push(name); }
    if (type === "Time In" && status === "LATE") late++;
    if ((values[i][3] || "").toString().trim() === "Guest") guests++;
    if (type === "Time Out") timeouts++;
  }
  return { present: present, late: late, outside: guests, timeouts: timeouts };
}

function updateMonthlySheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet(), now = new Date(), tz = Session.getScriptTimeZone();
  var sheetName = Utilities.formatDate(now, tz, "MMMM-yyyy"), sheet = ss.getSheetByName(sheetName);

  if (!sheet) sheet = ss.insertSheet(sheetName);
  sheet.clearContents(); sheet.clearFormats();
  
  buildTable1_Today(sheet, now, tz); 
  buildTable2_Month(sheet, now, tz);
}

function buildTable1_Today(sheet, now, tz) {
  var allData = getOrCreateSheet().getDataRange().getValues(), todayStr = Utilities.formatDate(now, tz, "yyyy-MM-dd");
  var entries = [], seenIds = {};

  for (var i = 1; i < allData.length; i++) {
    if (Utilities.formatDate(new Date(allData[i][0]), tz, "yyyy-MM-dd") !== todayStr || allData[i][5] !== "Time In") continue;
    var uid = allData[i][1], name = allData[i][2], cat = allData[i][3], status = allData[i][8];
    
    if (cat === "Guest") entries.push({ type: "guest", uniqueId: "", name: name, isLate: false });
    else if (!seenIds[uid]) { seenIds[uid] = true; entries.push({ type: "staff", uniqueId: uid, name: name, isLate: (status === "Late") }); }
  }

  sheet.getRange("A1:E1").merge().setValue("TODAY'S ATTENDANCE — " + Utilities.formatDate(now, tz, "MMMM dd, yyyy"))
       .setBackground("#1e3a8a").setFontColor("white").setFontWeight("bold").setHorizontalAlignment("center");
  sheet.getRange(2, 1, 1, 5).setValues([["#", "Unique ID", "Present", "Late", "Guest"]])
       .setBackground("#38bdf8").setFontColor("white").setFontWeight("bold");

  if (entries.length === 0) sheet.getRange(3, 1, 1, 5).merge().setValue("No attendance yet today.").setFontColor("#94a3b8").setHorizontalAlignment("center");

  for (var r = 0; r < entries.length; r++) {
    var rowNum = r + 3, entry = entries[r];
    sheet.getRange(rowNum, 1, 1, 5).clearContent().clearFormat();
    sheet.getRange(rowNum, 1).setValue(r + 1);

    if (entry.type === "staff") {
      sheet.getRange(rowNum, 2).setValue(entry.uniqueId); sheet.getRange(rowNum, 3).setValue(entry.name);
      sheet.getRange(rowNum, 4).setValue(entry.isLate ? entry.name : "");
      sheet.getRange(rowNum, 1, 1, 5).setBackground(r % 2 === 0 ? "#f0f4ff" : "#ffffff").setFontColor("#1e293b");
      if (entry.isLate) sheet.getRange(rowNum, 4).setFontColor("#dc2626").setFontWeight("bold");
    } else {
      sheet.getRange(rowNum, 5).setValue(entry.name);
      sheet.getRange(rowNum, 1, 1, 5).setBackground("#fffbeb");
      sheet.getRange(rowNum, 1).setFontColor("#92400e").setFontWeight("bold");
      sheet.getRange(rowNum, 5).setFontColor("#92400e").setFontWeight("bold");
    }
  }
  
  sheet.getRange(1, 1, Math.max(entries.length, 1) + 2, 5).setBorder(true,true,true,true,true,true,"#cbd5e1",SpreadsheetApp.BorderStyle.SOLID);
  
  // FIX: Properly setting individual column widths so the script doesn't crash
  sheet.setColumnWidth(1, 40);
  sheet.setColumnWidth(2, 80);
  sheet.setColumnWidth(3, 160);
  sheet.setColumnWidth(4, 160);
  sheet.setColumnWidth(5, 160);
}

function buildTable2_Month(sheet, now, tz) {
  var allData = getOrCreateSheet().getDataRange().getValues(), monthStr = Utilities.formatDate(now, tz, "yyyy-MM");
  var totalWorkingDays = countWorkingDays(now, tz), peopleMap = {};

  for (var i = 1; i < allData.length; i++) {
    if (Utilities.formatDate(new Date(allData[i][0]), tz, "yyyy-MM") !== monthStr || allData[i][5] !== "Time In") continue;
    var uid = allData[i][1];
    if (allData[i][3] === "Guest" || uid === "N/A") continue;

    if (!peopleMap[uid]) peopleMap[uid] = { name: allData[i][2], category: allData[i][3], presents: 0, lates: 0 };
    peopleMap[uid].presents++; if (allData[i][8] === "Late") peopleMap[uid].lates++;
  }

  var start = sheet.getLastRow() + 3;
  sheet.getRange(start, 1, 1, 6).merge().setValue("MONTHLY SUMMARY — " + Utilities.formatDate(now, tz, "MMMM yyyy"))
       .setBackground("#1e3a8a").setFontColor("white").setFontWeight("bold").setHorizontalAlignment("center");
  sheet.getRange(start + 1, 1, 1, 6).merge().setValue("Working days so far this month: " + totalWorkingDays)
       .setBackground("#0f172a").setFontColor("#94a3b8").setFontSize(10).setHorizontalAlignment("center");
  sheet.getRange(start + 2, 1, 1, 6).setValues([["Unique ID", "Name", "Category", "Present", "Late", "Total Working Days"]])
       .setBackground("#38bdf8").setFontColor("white").setFontWeight("bold");

  var keys = Object.keys(peopleMap);
  if (keys.length === 0) { sheet.getRange(start + 3, 1).setValue("No data yet for this month."); return; }

  for (var k = 0; k < keys.length; k++) {
    var uid = keys[k], p = peopleMap[uid], rNum = start + 3 + k;
    sheet.getRange(rNum, 1, 1, 6).setValues([[uid, p.name, p.category, p.presents, p.lates, totalWorkingDays]]);
    if (k % 2 === 0) sheet.getRange(rNum, 1, 1, 6).setBackground("#f0f4ff");
    if (p.lates > 0) sheet.getRange(rNum, 5).setFontColor("#dc2626").setFontWeight("bold");
    sheet.getRange(rNum, 4).setFontColor("#16a34a").setFontWeight("bold");
  }
  
  sheet.getRange(start, 1, keys.length + 3, 6).setBorder(true,true,true,true,true,true,"#cbd5e1",SpreadsheetApp.BorderStyle.SOLID);
  sheet.setColumnWidth(6, 160);
}

function countWorkingDays(now, tz) {
  var year = now.getFullYear(), month = now.getMonth(), today = now.getDate(), count = 0;
  for (var d = 1; d <= today; d++) { var day = new Date(year, month, d).getDay(); if (day !== 0 && day !== 6) count++; }
  return count;
}
